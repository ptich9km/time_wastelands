1. ТИПЫ (server/game/battle/state.ts) — Zod + TS:

type HexCoord = { q: number; r: number };  // axial coords, pointy-top
  q: int (-50..50), r: int (-50..50)

type Position = 'stand' | 'crouch' | 'run' | 'cover';

type ParticipantKind = 'player' | 'mob';

type Stats6 = { str, agi, int, end, per, intel } — все int >= 0 (у игрока per/intel могут быть 0; у моба уже провалидировано >= 1 в mobs.json)

type Equipment = {
  weaponId: string | null;   // null = без оружия
  armor: { head: string | null; torso: string | null; arms: string | null; legs: string | null; };
  // arms — экипировка рук (перчатки/наручи) — в items.json слот "arms" пока НЕ используется, но поле в state есть для совместимости
}

type Effect = {
  id: string;
  kind: 'poison' | 'bleed' | 'stun' | 'buff' | 'debuff';
  ttlRounds: int >= 0;
  payload?: Record<string, unknown>;  // произвольные данные (v0.2)
}

type Participant = {
  id: string;                      // уникальный в бою: "player" | "mob_1" | "mob_2"
  kind: ParticipantKind;
  refId: string;                   // characterId ("1") для игрока, mobId для моба
  stats: Stats6;
  hp: int >= 0;
  hpMax: int >= 1;
  ap: int >= 0;
  apMax: int >= 0;
  hex: HexCoord;
  position: Position;
  facing: int 0..5;                // v0.2, но поле есть
  equipment: Equipment;
  effects: Effect[];
  // для мобов:
  mobLevel?: int >= 1;
  mobNameKey?: string;
  // для игрока:
  characterName?: string;
}

type Obstacle = {
  hex: HexCoord;
  type: 'wall' | 'rock' | 'tree' | 'crate';
  blocksLos: boolean;              // блокирует линию огня
  blocksMove: boolean;             // блокирует движение
}

type BattleMap = {
  width: int >= 1;                 // в гексах
  height: int >= 1;
  obstacles: Obstacle[];
}

type BattleLogEntry = {
  round: int >= 1;
  actor: string;                   // "player" | "mob_1" | "system"
  type: 'hit' | 'miss' | 'crit' | 'info' | 'system' | 'flee' | 'death' | 'heal';
  text: string;                    // уже переведённый текст (не i18n-ключ)
  // ВАЖНО: лог пишется на языке клиента на момент события, чтобы при реконнекте не пересчитывать.
  // Если клиент на другом языке — лог не переводится (это by design, TZ так и делала).
  createdAt: string;               // ISO
}

type BattleContext = {
  locationId: string;
  isAmbush: boolean;
  cover: boolean;                  // локация с cover:true
  timeOfDay: 'day' | 'night';      // пока 'day', v0.2
}

type BattleStatus = 'active' | 'victory' | 'defeat' | 'fled';

type BattleState = {
  id: string;                      // uuid v4
  round: int >= 1;
  status: BattleStatus;
  phase: 'player_turn' | 'enemy_turn' | 'finished';
  map: BattleMap;
  participants: Participant[];     // минимум 2 (игрок + 1 моб)
  turnOrder: string[];             // id в порядке инициативы (пока: ["player", "mob_1", ...])
  currentActorId: string;
  log: BattleLogEntry[];           // append-only, в БД отдельная таблица
  context: BattleContext;
  healLimit: { current: int >= 0; max: int >= 0; regenPerRound: int >= 0 };
  result?: 'victory' | 'defeat' | 'fled';
  createdAt: string;
  updatedAt: string;
}

Zod-схемы: BattleStateSchema, ParticipantSchema, BattleLogEntrySchema, и т.д. Валидация hex.q/r в диапазоне -50..50.

ВАЖНО: Zod-схема — источник правды. Если поле добавится в v0.2 — расширяй схему с .optional() с default.

2. СХЕМА БД (server/db/schema.sql, CREATE TABLE IF NOT EXISTS):
   battle_sessions (
     id TEXT PRIMARY KEY,                       -- uuid v4
     character_id INTEGER NOT NULL DEFAULT 1 CHECK (character_id = 1),
     status TEXT NOT NULL CHECK (status IN ('active', 'victory', 'defeat', 'fled')),
     state_json TEXT NOT NULL,                  -- JSON BattleState БЕЗ log (лог отдельно)
     created_at TEXT NOT NULL DEFAULT (datetime('now')),
     updated_at TEXT NOT NULL DEFAULT (datetime('now')),
     FOREIGN KEY (character_id) REFERENCES characters(id) ON DELETE CASCADE
   )
   -- Индекс: (character_id, status) — для getActiveBattleForCharacter
   -- Частичный unique: CREATE UNIQUE INDEX ... ON battle_sessions(character_id) WHERE status = 'active'
   --   (у одного персонажа не может быть двух активных боёв)

   battle_log (
     id INTEGER PRIMARY KEY AUTOINCREMENT,
     battle_id TEXT NOT NULL,
     round INTEGER NOT NULL,
     actor TEXT NOT NULL,
     type TEXT NOT NULL CHECK (type IN ('hit','miss','crit','info','system','flee','death','heal')),
     text TEXT NOT NULL,
     created_at TEXT NOT NULL DEFAULT (datetime('now')),
     FOREIGN KEY (battle_id) REFERENCES battle_sessions(id) ON DELETE CASCADE
   )
   -- Индекс: (battle_id, id)

3. РЕПОЗИТОРИЙ (server/db/battle-repo.ts):
   - createBattle(db, state: BattleState) — вставить в battle_sessions (state_json без поля log), в battle_log вставить все записи из state.log. Транзакционно.
   - getBattle(db, id) — вернуть BattleState | null: state_json + собрать log из battle_log ORDER BY id.
   - updateBattleState(db, state) — UPDATE state_json, updated_at; новые записи log — append (не перезаписывать). ВАЖНО: log в state — это ВСЁ, что было + новое. При update — сравнивать длину с уже сохранённым логом и добавлять только новые. (Или: appendLog отдельно. Выбери подход — но без дублей и без потери.)
   - appendLog(db, battleId, entries: BattleLogEntry[]) — добавить пачку записей.
   - getActiveBattleForCharacter(db) — вернуть BattleState | null (status='active').
   - markFinished(db, id, result) — status='victory'|'defeat'|'fled', updated_at.
   - Все — prepared statements, транзакции где нужно.

4. REST API (server/api/battle.ts — новый файл, регистрируется в index.ts):
   - POST /api/battle/start → body { mobIds: string[] } — минимально 1 моб, максимально 3.
     * Валидация: персонаж есть; mobIds существуют в gameData.mobs; у персонажа нет активного боя (409 battle_already_active); все мобы в сумме level <= character.level + 5 (иначе 400 too_hard); локация персонажа — в мире.
     * Создать BattleState:
       - map: 10×10, obstacles: 3–4 случайных препятствия (wall/rock/tree/crate), не на гексах спавна. Если в world.json у локации есть battleMap — использовать его (пока нет — дефолт).
       - participants: player (из character + equipped из inventory) + по одному mob на каждый mobId. Позиции: player в (2, 5), мобы в (7, 5), (7, 4), (7, 6) — не на препятствиях, при конфликте — сдвиг.
       - stats: player из character; mob из mob.stats.
       - hp/hpMax: player из character.hp_current/hp_max; mob из mob.hp.
       - ap/apMax: player = ap_max; mob = 4 (фиксировано, mob.moveAp в MVP не используется).
       - position: 'stand' для всех; facing: 0 (на восток — v0.2).
       - equipment: player — надетое; mob — { weaponId: null, armor: { head:null,torso:null,arms:null,legs:null } } (броня мобов — v0.2).
       - healLimit: { current: floor(hpMax * 1.5), max: floor(hpMax * 1.5), regenPerRound: floor(hpMax * 0.2) }.
       - turnOrder: ['player', ...mobIds], currentActorId = 'player'.
       - log: [{ round: 1, actor: 'system', type: 'system', text: t('battle.start', lang, { target: mobName }) }] — стартовая запись.
       - context: { locationId: character.location_id, isAmbush: false, cover: world.location.cover ?? false, timeOfDay: 'day' }.
     * Сохранить в БД.
     * 201 { battleId, state } (state с log).
   - GET /api/battle/:id → 200 { state } | 404 { error: 'battle_not_found' }.
     * Возвращать state с логом.
   - GET /api/battle/active → 200 { state } | 404 { error: 'no_active_battle' }.
   - Язык лога: из query ?lang= или из Accept-Language или из тела (пока — из query ?lang=ru|en, default ru). Клиент передаёт.
