3. ФОРМУЛЫ (server/game/battle/formulas.ts) — чистые функции с инъекцией RNG:

type Rng = () => number;  // 0 <= rng() < 1

// --- Hit ---
function hitChance(params: {
  perAttacker: number;
  agiTarget: number;
  zone: 'head'|'torso'|'arms'|'legs';
  distanceHexes: number;       // расстояние в гексах (>= 0; 0 = тот же гекс)
  weapon: { range: number; spread: number; sniper: boolean };
  positionAttacker: Position;  // stand|crouch|run|cover
  positionTarget: Position;
  coverTarget: boolean;        // цель в cover
  config: HitConfig;
}): { chance: number; distanceMod: number; zoneMod: number; positionMod: number; coverMod: number; pointBlank: boolean }
  - pointBlank = distanceHexes <= config.pointBlankHexes
  - Если pointBlank И positionAttacker in ('stand','crouch') И !coverTarget И positionTarget in ('stand','crouch') → chance = 100 (clamp не применяется)
  - Иначе:
    * baseRatio = perAttacker / max(agiTarget, 1)
    * zoneMod = baseRatio >= 1 ? config.zoneModsIfPerAboveAgi : config.zoneMods[zone]
    * distanceMod: линейно от 1.0 (1 гекс) до (weapon.spread / 100) на (weapon.range) гексах; за range — (weapon.spread / 100) (не падает дальше). Для sniper: 0.1 на 1 гексе → (spread/100) на (0.7 × range) → плато.
      ВАЖНО: distanceHexes = 0 (тот же гекс) → distanceMod = 1.0.
    * positionMod = 1 + (config.positionMods[positionAttacker].selfPenalty + config.positionMods[positionTarget].enemyPenalty) / 100
    * coverMod = coverTarget ? 1 + config.coverBonus / 100 : 1.0
    * chance = 100 × baseRatio × zoneMod × distanceMod × positionMod × coverMod
    * clamp chance в [clampMin, clampMax]
  - Вернуть объект с chance и компонентами (для лога и UI).

// --- Damage ---
function rollDamage(params: {
  weaponDamage: { min: number; max: number };
  ammoBonus: number;           // 0, если нет патронов
  rng: Rng;
}): number
  - randInt(min, max) через rng
  - + ammoBonus

function rollCrit(params: {
  intel: number;
  config: DamageConfig;
  rng: Rng;
}): boolean
  - chance = (config.critChanceBase + intel × config.critChancePerIntel) / 100
  - return rng() < chance

function applyCrit(damage: number, config: DamageConfig): number
  - damage × config.critMultiplier (round)

// --- Armor ---
function armorAbsorb(params: {
  armorValue: number;          // Y
  weaponLevel: number;         // X
  config: ArmorConfig;
}): number  // процент 0..maxAbsorb
  - if weaponLevel <= 6:
      absorb = min(armorValue × config.formulaUpTo6.mult50 / config.formulaUpTo6.cap, config.maxAbsorb)
    else:
      absorb = min(armorValue × config.formulaAbove6.mult50 / (config.formulaAbove6.coef × weaponLevel - config.formulaAbove6.offset), config.maxAbsorb)
  - ВАЖНО: знаменатель может быть <= 0 (при coef×X - offset <= 0) → absorb = maxAbsorb (защита «бесконечна»)
  - return clamp(absorb, 0, maxAbsorb)

function damageAfterArmor(params: {
  damage: number;              // base после crit, ДО брони
  absorb: number;              // процент 0..maxAbsorb
}): number
  - final = damage × (1 - absorb / 100)
  - return max(round(final), config.minDamage)

// --- Penetration ---
function rollPenetration(params: {
  pen: number;                 // 0..100
  rng: Rng;
}): boolean
  - return rng() < pen / 100

function damageWithPenetration(params: {
  baseDamage: number;          // до брони, после crit
  absorb: number;              // процент
  penetrated: boolean;
  config: PenetrationConfig;
}): number
  - если !penetrated → обычный урон: base × (1 - absorb/100)
  - если penetrated:
    * absorb < fullDamageBelow (40) → base × (1 - absorb/100) + base × (absorb/100) = base  (полный урон)
    * absorb <= linearTo (70) → обычный + бонус_бронебоя × (1 - penalty), где:
        ordinary = base × (1 - absorb/100)
        bonus = base - ordinary     (= base × absorb/100)
        penalty = (absorb - linearFrom) / (linearTo - linearFrom)
        final = ordinary + bonus × (1 - penalty)
    * absorb > ignoredAbove (70) → обычный урон (бронебой игнорируется)
  - return max(round(final), config.minDamage)

// --- Distance (для урона вне range) ---
function damageOutsideRange(params: {
  damage: number;
  distanceHexes: number;
  weaponRange: number;
  config: DistanceConfig;
}): number
  - если distanceHexes <= weaponRange → damage
  - иначе: damage / (config.outsideRangeDivisor ^ (distanceHexes - weaponRange))
  - return max(round, minDamage)

// --- XP ---
function xpForKill(params: {
  mobHp: number;
  mobLevel: number;
  playerLevel: number;
  config: XpConfig;
}): number
  - diff = mobLevel - playerLevel
  - mod = clamp(1 + diff × config.levelDiffMod, config.levelDiffClampMin, config.levelDiffClampMax)
  - return floor(mobHp × config.perHpBase × mod)

function xpToNextLevel(params: {
  level: number;
  config: XpConfig;
}): number
  - return config.basePerLevel × level

// --- Level up ---
function applyLevelUp(params: {
  character: { str; agi; int; end; per; intel; hp; hpMax; apMax; level };
  newLevel: number;
  levelTable: LevelTable;
}): { character, leveled: boolean, oldLevel: number }
  - если newLevel <= character.level → leveled: false, вернуть как есть
  - взять строку из levelTable по newLevel
  - обновить статы, hpMax = row.hpMax, apMax = row.apMax
  - hp: ВАЖНО — при левелапе HP ВОССТАНАВЛИВАЕТСЯ до hpMax (по TZ; если не так — поправим конфигом; в MVP — восстанавливаем)
  - вернуть
